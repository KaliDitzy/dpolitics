on_actions = {
    on_startup = {
        effect = {
            every_possible_country = {
                recalculate_ideological_incoherence = yes
                recalculate_dissent = yes
                randomize_ideology_targets = yes
                recalculate_gui = yes
                recalculate_cosmetic_tag = yes

                d_spawn_army = yes
            }
        }
    }
    on_daily = {
        effect = {
            recalculate_cooldowns = yes

            recalculate_ideological_incoherence = yes
            recalculate_dissent = yes
            recalculate_gui = yes
            recalculate_cosmetic_tag = yes

            set_variable = { global.day_1_over = 1 }
        }
        effect = {
            # REGIME APPROVAL
            if = {
                limit = {
                    check_variable = { regime_approval < 0.1 }
                    check_variable = { regime_approval > -0.1 }
                }
                set_variable = { regime_approval = 0 }
            }
            else_if = {
                limit = {
                    check_variable = { regime_approval > 0 }
                }
                subtract_from_variable = { regime_approval = 0.1 }
            }
            else_if = {
                limit = {
                    check_variable = { regime_approval < 0 }
                }
                add_to_variable = { regime_approval = 0.1 }
            }
        }
    }
    on_weekly = {
        effect = { # regular stuff
            add_to_variable = { dissent_base = dissent_change }
            recalculate_dissent = yes
            
            update_ideology_targets = yes
        }
        effect = { # random scandal
            set_temp_variable_to_random = t_rand
            multiply_temp_variable = { t_rand = 208 }
            multiply_temp_variable = { t_rand = stability }
            clamp_temp_variable = {
                var = t_rand
                min = 36
            }
            if = {
                limit = {
                    check_variable = { t_rand < 9 } # 1
                }
                random_list = {
                    100 = {
                        modifier = {
                            factor = party_popularity@fascism
                        }
                        country_event = politics.scandal.1
                    }
                    100 = {
                        modifier = {
                            factor = party_popularity@traditionalism
                        }
                        country_event = politics.scandal.2
                    }
                    100 = {
                        modifier = {
                            factor = party_popularity@conservatism
                        }
                        country_event = politics.scandal.3
                    }
                    100 = {
                        modifier = {
                            factor = party_popularity@liberalism
                        }
                        country_event = politics.scandal.4
                    }
                    100 = {
                        modifier = {
                            factor = party_popularity@radicalism
                        }
                        country_event = politics.scandal.5
                    }
                    100 = {
                        modifier = {
                            factor = party_popularity@social_democracy
                        }
                        country_event = politics.scandal.6
                    }
                    100 = {
                        modifier = {
                            factor = party_popularity@democratic_socialism
                        }
                        country_event = politics.scandal.7
                    }
                    100 = {
                        modifier = {
                            factor = party_popularity@bolshevism
                        }
                        country_event = politics.scandal.8
                    }
                    100 = {
                        modifier = {
                            factor = party_popularity@anarchism
                        }
                        country_event = politics.scandal.9
                    }
                }
            }
        }
        random_events = {
            100 = politics.border_influence.1
            100 = politics.border_influence.2
            100 = politics.border_influence.3
            100 = politics.border_influence.4
            100 = politics.border_influence.5
            100 = politics.border_influence.6
            100 = politics.border_influence.7
            100 = politics.border_influence.8
            100 = politics.border_influence.9
        }
    }
    on_monthly = {
        effect = {
            add_to_variable = {
                current_month = 1
            }
            if = {
                limit = {
                    check_variable = { current_month > 11 }
                }
                set_variable = { current_month = 0 }
            }
            if = {
                limit = {
                    check_variable = { current_month = 0 }
                }
                randomize_ideology_targets = yes
            }
        }
    }
}
